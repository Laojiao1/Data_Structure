# 1. Tree Traversals Again

### 一、题意理解

题目描述：
我们可以使用堆栈以非递归的方式实现中序二叉树遍历。例如，假设我们现在中序遍历一棵包含6个结点的二叉树(编号为1到6)，对应堆栈的操作为：push(1); push(2); push(3); pop(); pop(); push(4); pop(); pop(); push(5); push(6); pop(); pop()。根据这个操作步骤，我们可以生成一个唯一的二叉树(如下图所示)。你的任务是给出这棵树的后序遍历序列。

输入格式：
输入的第一行为一个整数N(≤30)，它代表树的结点的总数(结点的编号从1到N)。后面的2N行每行描述一个堆栈操作，格式为：Push X代表将结点X压入堆栈，Pop表示从堆栈中弹出一个结点。

输出格式：
在一行中打印对应树的后序遍历序列，所有结点编号用一个空格隔开，并且行尾不能有多余的空格。

输入样例：

```
6
Push 1
Push 2
Push 3
Pop
Pop
Push 4
Pop
Pop
Push 5
Push 6
Pop
Pop
```

输出样例：

```
3 4 2 6 5 1
```

### 三、整体代码

```c
#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
#include<stdlib.h>
#include <string.h>

typedef struct SNode *Stack;
struct SNode {
	int *Data;
	int Top;
};

void Push(Stack PtrS, int item, int MaxSize)
{
	if (PtrS->Top == MaxSize - 1) {
		printf("堆栈满"); 
		return;
	}
	else {
		PtrS->Data[++(PtrS->Top)] = item;
		return;
	}
}

int Pop(Stack PtrS)
{
	if (PtrS->Top == -1) {
		printf("堆栈空");
		return -1; /* ERROR是ElementType的特殊值，标志错误 */
	}
	else
		return (PtrS->Data[(PtrS->Top)--]);
}

//preL代表先序遍历最左边的元素
//inL代表中序遍历最左边的元素
//postL代表后序遍历最左边的元素
//n代表元素个数
void solve(int preL, int inL, int postL, int n, int *pre, int *in, int *post)
{
	int root;
	int L,R;
	int i = 0;
	//n=0时需要返回，比如上面我们最后递归调用到5这个结点，它没有右子树，但是程序不知道，这个时候应该直接返回
	if (n == 0) return;
	//n=1，说明没有子树不需要递归了
	if (n == 1) { post[postL] = pre[preL]; return; }
	//先序遍历最左边的元素是根结点
	root = pre[preL];
	//将根结点放到后序遍历的最后面
	post[postL + n - 1] = root;
	//去中序遍历数组中找根结点在哪里
	for (i = 0; i < n; i++)
		if (in[inL + i] == root) break;
	//左子树结点个数为下标i
	//右子树结点个数为n-L-1，因为要减去这个根结点
	L = i; R = n - L - 1;
	/*
	递归调用左子树
	preL最开始是结点1，现在要向右移动到结点2
	inL和postL最开始是结点3，现在不动
	左子树长度就是前面算出来的N
	*/
	solve(preL + 1, inL, postL, L, pre, in, post);
	/*
	递归调用右子树
	preL最开始是结点1，现在要向右移动到结点5,移动L+1个位置
	inL最开始是结点3，现在要向右移动到结点6,移动L+1个位置
	postL最开始是结点3，现在要向右移动到结点5,移动L个位置
	右子树长度就是前面算出来的R
	*/
	solve(preL + L + 1, inL + L + 1, postL + L, R, pre, in, post);
}

int main()
{
	int num;     //结点总数
	int value;   //push操作时读入的数
	scanf("%d", &num);
	//创建前序、中序、后序三个数组
	int *pre = (int*)malloc(sizeof(int)*num);
	int *in = (int*)malloc(sizeof(int)*num);
	int *post = (int*)malloc(sizeof(int)*num);

	//创建堆栈，Top初始值微-1，代表空栈
	Stack S = (Stack)malloc(sizeof(struct SNode));
	S->Top = -1;
	S->Data = (int *)malloc(sizeof(int)*num); 

	char word[8];   //读入数据
	int pre_i = 0; //控制将数据存入前序数组的位置
	int in_i = 0;  //控制将数据存入中序数组的位置

	for (int i = 0; i < num * 2; i++)
	{
		scanf("%s", word);
		if (strcmp(word, "push") == 0)
		{
			scanf("%d", &value);
			Push(S, value, num);
			pre[pre_i] = value;
			pre_i++;
		}
		else 
		{
			in[in_i] = Pop(S);
			in_i++;
		}
	}

	solve(0, 0, 0, num, pre, in, post);
	printf("\n");
	for (int l = 0; l < num; l++)
	{
		printf("%d ", post[l]);
	}
	return 0;
}
```
