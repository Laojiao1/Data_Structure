# 二逼平衡树 — 树套树

**树套树：线段树套平衡树**

线段树的每个结点对应一个区间，在每个节点上构建一颗平衡树，维护区间内的序列。

平衡树可以用Splay、FHQ Treap

本篇运用Splay

## 进入正题

### 准备工作

```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 50005;
#define INF 2147483647
#define ls(x) t[x].s[0]
#define rs(x) t[x].s[1]
#define lc u << 1
#define rc u << 1 | 1
struct Node {
    int s[2];
	int p;
    int size;
    int v;
    void init(int p1, int v1) {
        p = p1;
        v = v1;
        size = 1;
    }
} t[N*50];
int n, m, w[N], idx;
int root[N*4]; // 记录每个splay的根
inline void push_up(int x) {
    t[x].size = t[ls(x)].size + t[rs(x)].size + 1;
}
```

### Splay

```c++
inline void rotate(int x) {
    int y = t[x].p;
   	int z = t[y].p;
    int k = t[y].s[1] == x;
    
    t[y].s[k] = t[x].s[k^1];
    t[t[x].s[k^1]].p = y;
    
    t[x].s[k^1] = y;
    t[y].p = x;
    
    t[z].s[t[z].s[1] == y] = x;
    t[x].p = z;
    
    push_up(y);
    push_up(x);
}
inline void splay(int &root, int x, int k) {
    while(t[x].p != k) {
        int y = t[x].p;
        int z = t[y].p;
        if(z != k) {
            (rs(y) == x) ^ (rs(z) == y) ? rotate(x) : rotate(y);
        }
        rotate(x);
    }
    if(k == 0) root = x;
}
```

### 树套树的构建

外层线段树正常构建，内层平衡树在每个节点上构建

例如：插入 5 2 8 6 8 6

```c++
inline void insert(int& root, int v) {
    int u = root, p = 0;
    while(u) {
        p = u;
        u = t[u].s[v > t[u].v];
    }
    u = ++idx;
    t[p].s[v > t[p].v] = u;
    t[u].init(p, v);
    splay(root, u, 0);
}
void build(int u, int l, int r) {
    insert(root[u], -INF), insert(root[u], INF);
    for(int i = l; i <= r; i++) {
        insert(root[u], w[i]);
    }
    if(l == r) return;
    int mid = (l + r) >> 1;
    build(lc, l, mid);
    build(rc, mid + 1, r);
}
```

### 求某区间中某值的排名

线段树负责裂开，平衡树负责查找。

把线段树不断裂开，遇到已覆盖的区间，在该区间平衡树中查找比某值小的元素个数，区间结果合并时，将小的元素个数求和。

线段树走logn层，平衡树查logn层，所以每次时间复杂度为log^2n

例如：x = 2, y = 5, v = 8

<img src="C:\Users\44174\AppData\Roaming\Typora\typora-user-images\image-20220821154239587.png" alt="image-20220821154239587" style="zoom:33%;" />

```c++
int getrank(int root, int v) {
    int u = root, res = 0;
    while(u) {
       	if(t[u].v < v) {
            res += t[ls(u)].size + 1;
            u = rs(u);
        } else {
            u = ls(u);
        }
    }
    return res;
}
int queryrank(int u, int l, int r, int x, int y, int v) {
    if(x <= l && r <= y) return getrank(root[u], v) - 1;
    int mid = (l + r) >> 1, res = 0;
    if(x <= mid) res += queryrank(lc, l, mid, x, y, v);
    if(y > mid) res += queryrank(rc, mid+1, r, x, y, v);
    return res;
}
```

### 求某区间中排名为k的数值

二分答案求排名

```c++
int queryval(int u, int x, int y, int k) {
    int l = 0, r = 1e8, ans;
    while(l <= r) {
        int mid = (l + r) >> 1;
        if(queryrank(1, 1, n, x, y, mid) + 1 <= k) {
            l = mid+1;
            ans = mid;
        } else {
            r = mid - 1;
        }
    }
    return ans;
}
```

### 修改某一位置上的数值

沿着线段树的路径走，穿透每一层，在所有包含树的平衡树中先删除某树，然后再插入另一个树

<img src="C:\Users\44174\AppData\Roaming\Typora\typora-user-images\image-20220821160647619.png" alt="image-20220821160647619" style="zoom:33%;" />

```c++
void change(int u, int l, int r, int pos, int v) {
    del(root[u], w[pos]);
    insert(root[u], v);
    if(l == r) return;
    int mid = (l+r) >> 1;
    if(pos <= mid) change(lc, l, mid, pos, v);
    else change(rc, mid+1, r, pos, v);
}
inline void del(int& root, int v) {
    int u = root;
    while(u) {
        if(t[u].v == v) break;
        if(t[u].v < v) u = rs(u);
        else u = ls(u);
	}
    int l = ls(u), r = rs(u);
    while(rs(l)) l = rs(l);
    while(ls(r)) r = ls(r);
    splay(root, l, 0);
    splay(root, r, l);
    ls(r) = 0;
    splay(root, r, 0);
}
```

### 求某区间中某值的前驱

把线段树不断裂开，遇到已覆盖的区间，在该区间平衡树中查找某值的前驱，区间结果合并时，**取最大值。**

```c++
int querypre(int u, int l, int r, int x, int y, int v) {
    if(x <= l && r <= y) return getpre(root[u], v);
    int mid = (l + r) >> 1, res = -INF;
    if(x <= mid) res = max(res, querypre(lc, l, mid, x, y, v));
    if(y > mid) res = max(res, querypre(rc, mid+1, r, x, y, v));
    return res;
}
inline int getpre(int root, int v) {
    int u = root, res = -INF;
    while(u) {
        if(t[u].v < v) {
            res = t[u].v;
            u = rs(u);
        } else {
            u = ls(u);
        }
    }
    return res;
}
```
